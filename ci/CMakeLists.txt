cmake_minimum_required(VERSION 3.22)
project(Tibia860Android LANGUAGES C CXX)
if(NOT ANDROID OR NOT ANDROID_ABI STREQUAL "arm64-v8a")
  message(FATAL_ERROR "This target requires Android arm64-v8a")
endif()
set(CMAKE_CXX_STANDARD 17)
get_filename_component(OTCV8_ROOT "${CMAKE_CURRENT_LIST_DIR}/.." ABSOLUTE)
include(android-sources.cmake)
add_library(otclientv8 SHARED ${OTCV8_SOURCES})
target_include_directories(otclientv8 PRIVATE "${OTCV8_ROOT}/src" "${ANDROID_DEPS}/include")
target_compile_definitions(otclientv8 PRIVATE ANDROID OPENGL_ES OTCLIENT FW_GRAPHICS FW_NET FW_XML FW_PROXY NOMINMAX NDEBUG _LIBCPP_ENABLE_CXX17_REMOVED_UNARY_BINARY_FUNCTION)
target_precompile_headers(otclientv8 PRIVATE "${OTCV8_ROOT}/src/android/pch.h")
target_compile_options(otclientv8 PRIVATE -O2 -fexceptions -frtti -Wno-deprecated-declarations)
target_link_options(otclientv8 PRIVATE -Wl,-z,max-page-size=16384 -Wl,-z,common-page-size=16384)
set(deps boost_system vorbisfile vorbis ssl png physfs openal ogg luajit crypto zip)
set(archives)
foreach(dep IN LISTS deps)
  if(NOT EXISTS "${ANDROID_DEPS}/lib64/lib${dep}.a")
    message(FATAL_ERROR "Missing ARM64 dependency: ${dep}")
  endif()
  list(APPEND archives "${ANDROID_DEPS}/lib64/lib${dep}.a")
endforeach()
target_link_libraries(otclientv8 PRIVATE -Wl,--start-group ${archives} -Wl,--end-group android log EGL GLESv2 z dl)
